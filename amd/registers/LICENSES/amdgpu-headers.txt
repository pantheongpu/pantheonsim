asic_reg/mp/mp_13_0_6_offset.h, asic_reg/nbio/nbio_7_9_0_offset.h,
pm/swsmu/inc/pmfw_if/smu_v13_0_6_ppsmc.h and include/kgd_pp_interface.h --
