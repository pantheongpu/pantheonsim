# ---------------------------------------------------------------------------
# libvgpucudart: the CUDA *runtime* API shim. Lets unmodified nvcc-compiled
# applications (built against shared cudart) run on VirtualGPU. Requires the
# CUDA toolkit's ABI headers at build time (driver_types.h) for exact struct
# layout; built only when they are found, so the core stays dependency-free.
# find_path keeps a path it has cached even after that directory is gone, and
# never searches again. Every build directory configured before the vendored
# headers moved from third_party/ to nvidia/third_party/ kept the old path, and
# the build then failed with "nccl.h: No such file or directory". A cached path
# whose header no longer exists is forgotten, so the search runs again.
function(vgpu_forget_stale_path var header)
  if(${var} AND NOT EXISTS "${${var}}/${header}")
    message(STATUS "${var}: ${${var}}/${header} is gone; searching again")
    unset(${var} CACHE)
  endif()
endfunction()

# ---------------------------------------------------------------------------
# Derive the ABI include from the nvcc that will compile the workloads, so the
# shim's cudaDeviceProp layout matches the application's exactly. A mismatched
# system copy (e.g. an older /usr/include/driver_types.h) would corrupt caller
# structs, so nvcc's own toolkit include is the only trustworthy source.
find_program(VGPU_NVCC nvcc)
if(VGPU_NVCC)
  # Resolve symlinks first: distributions put an nvcc symlink in /usr/bin, and
  # deriving the toolkit root from that gives /usr, where no CUDA library lives.
  get_filename_component(_nvcc_real "${VGPU_NVCC}" REALPATH)
  get_filename_component(_nvcc_bin "${_nvcc_real}" DIRECTORY)
  get_filename_component(_cuda_root "${_nvcc_bin}" DIRECTORY)
  find_path(CUDA_ABI_INCLUDE driver_types.h
    PATHS "${_cuda_root}/include" "${_cuda_root}/targets/x86_64-linux/include"
    NO_DEFAULT_PATH)
endif()
if(NOT CUDA_ABI_INCLUDE)
  find_path(CUDA_ABI_INCLUDE driver_types.h
    PATHS /usr/local/cuda/include /opt/cuda/include ENV CUDA_HOME ENV CUDA_PATH
    PATH_SUFFIXES include NO_DEFAULT_PATH)
endif()
# The soname is what the loader matches on, and each CUDA library carries its
# own major version that moves between toolkits (cuBLAS is .so.12 under CUDA 12
# and .so.13 under CUDA 13). Building under the wrong one means programs
# silently load NVIDIA's real library instead of this one -- which is exactly
# what happened when cuRAND was built as .so.13. So rather than hard-code a
# table that goes stale, read the major off the toolkit that is actually
# installed, and fall back to the documented value if the library is absent.
function(vgpu_toolkit_soversion _name _default _out)
  set(_found "")
  # The include directory that was actually accepted is the most reliable
  # anchor: its sibling lib directories belong to the same toolkit.
  get_filename_component(_inc_root "${CUDA_ABI_INCLUDE}" DIRECTORY)
  # Distro-packaged toolkits (Ubuntu's cuda-* debs) put the headers in
  # /usr/include and the libraries in the multiarch dir, which none of the
  # toolkit-shaped paths above cover: the search fell through to the default
  # and built a .so.13 shim on a CUDA 12 host, where it is never loaded
  # because the program asks the loader for libcudart.so.12.
  foreach(_dir "${_cuda_root}/lib64" "${_cuda_root}/targets/x86_64-linux/lib" "${_cuda_root}/lib"
               "${_cuda_root}/lib/${CMAKE_LIBRARY_ARCHITECTURE}"
               "${_inc_root}/lib64" "${_inc_root}/lib"
               "${_inc_root}/lib/${CMAKE_LIBRARY_ARCHITECTURE}"
               "/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}" "/lib/${CMAKE_LIBRARY_ARCHITECTURE}")
    file(GLOB _cands "${_dir}/lib${_name}.so.[0-9]*")
    foreach(_c IN LISTS _cands)
      if("${_c}" MATCHES "\\.so\\.([0-9]+)$")
        set(_found "${CMAKE_MATCH_1}")
        break()
      endif()
    endforeach()
    if(_found)
      break()
    endif()
  endforeach()
  if(NOT _found)
    set(_found "${_default}")
  endif()
  set(${_out} "${_found}" PARENT_SCOPE)
endfunction()

# A version script naming the soname's own version node, which is how NVIDIA
# tags these symbols. Getting it wrong makes the loader warn about missing
# version information on every call.
function(vgpu_version_script _libname _soversion _out)
  set(_path "${CMAKE_BINARY_DIR}/maps/lib${_libname}.so.${_soversion}.map")
  file(WRITE "${_path}" "lib${_libname}.so.${_soversion} {\n  global: *;\n};\n")
  set(${_out} "${_path}" PARENT_SCOPE)
endfunction()

            "${CMAKE_BINARY_DIR}/shim/libnvidia-ml.so"
    COMMENT "Creating NVML shim symlinks (build/shim/libnvidia-ml.so.1)")

  # ---------------------------------------------------------------------------
  # libvgpucupti: CUPTI, which is how every profiler reads a CUDA program --
  # Nsight, nvprof, PyTorch's profiler, DCGM. Without it the counters and the
  # timeline this engine already has are reachable only through VGPU_COUNTERS,
  # which no existing tool knows how to read. See nvidia/docs/cupti.md for what the
  # timestamps do and do not mean.
  # ---------------------------------------------------------------------------
  vgpu_toolkit_soversion(cupti 13 SO_CUPTI)
  vgpu_version_script(cupti ${SO_CUPTI} _cupti_map)
  # The runtime-API callback IDs are the toolkit's, so the table of them is
  # read from its header here rather than kept by hand: {function, versioned
  # name, ID} for every one the header names. Without cupti.h beside it there is
  # no CUPTI to build, and the guard on the target below says so.
  set(_cbid_header "${CUDA_ABI_INCLUDE}/cupti_runtime_cbid.h")
  set(_cbid_inc "${CMAKE_BINARY_DIR}/gen/cupti_runtime_cbids.inc")
  set(_cbid_text "")
  if(EXISTS "${_cbid_header}")
    file(STRINGS "${_cbid_header}" _cbid_lines REGEX "CUPTI_RUNTIME_TRACE_CBID_[A-Za-z0-9_]+_v[0-9]+ *=")
    foreach(_line IN LISTS _cbid_lines)
      if(_line MATCHES "CUPTI_RUNTIME_TRACE_CBID_([A-Za-z0-9_]+)_v([0-9]+) *=")
        set(_fn "${CMAKE_MATCH_1}")
        set(_ver "${CMAKE_MATCH_2}")
        string(APPEND _cbid_text
               "VGPU_CBID_ROW(\"${_fn}\", \"${_fn}_v${_ver}\", CUPTI_RUNTIME_TRACE_CBID_${_fn}_v${_ver})\n")
      endif()
    endforeach()
  endif()
  file(WRITE "${_cbid_inc}.tmp" "${_cbid_text}")
  execute_process(COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_cbid_inc}.tmp" "${_cbid_inc}")
  # The driver-API ones, from cupti_driver_cbid.h. The names are not versioned
  # the runtime's way (cudaMalloc_v3020): they are the driver function's own
  # spelling, cuMemAlloc_v2 and cuMemAlloc being two functions with an ID each,
  # and the name a subscriber is told is exactly that. A second file says which
  # exist, because the toolkits differ (cuCtxCreate_v4 is newer than 12.0).
  set(_dcbid_header "${CUDA_ABI_INCLUDE}/cupti_driver_cbid.h")
  set(_dcbid_inc "${CMAKE_BINARY_DIR}/gen/cupti_driver_cbids.inc")
  set(_dcbid_have "${CMAKE_BINARY_DIR}/gen/cupti_driver_have.h")
  set(_dcbid_text "")
  set(_dcbid_have_text "")
  if(EXISTS "${_dcbid_header}")
    file(STRINGS "${_dcbid_header}" _dcbid_lines REGEX "CUPTI_DRIVER_TRACE_CBID_cu[A-Za-z0-9_]+ *=")
    foreach(_line IN LISTS _dcbid_lines)
      if(_line MATCHES "CUPTI_DRIVER_TRACE_CBID_(cu[A-Za-z0-9_]+) *=")
        set(_fn "${CMAKE_MATCH_1}")
        string(APPEND _dcbid_text "VGPU_DRIVER_CBID_ROW(\"${_fn}\", CUPTI_DRIVER_TRACE_CBID_${_fn})\n")
        string(APPEND _dcbid_have_text "#define VGPU_HAVE_DRIVER_CBID_${_fn} 1\n")
      endif()
    endforeach()
  endif()
  file(WRITE "${_dcbid_inc}.tmp" "${_dcbid_text}")
  execute_process(COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_dcbid_inc}.tmp" "${_dcbid_inc}")
  file(WRITE "${_dcbid_have}.tmp" "${_dcbid_have_text}")
  execute_process(COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_dcbid_have}.tmp" "${_dcbid_have}")
  add_library(vgpucupti SHARED nvidia/src/cupti_api.cpp)
  # Against the runtime shim rather than the static core, deliberately. The
  # profiling switch and its event queue are process-global state: linking the
  # static library here would give this library its own copy, so CUPTI would
  # turn recording on in one and the runtime would keep reading the other, and
  # a profiler would attach successfully and receive nothing.
  target_link_libraries(vgpucupti PRIVATE vgpucudart)
  target_include_directories(vgpucupti PRIVATE ${CUDA_ABI_INCLUDE} include "${CMAKE_BINARY_DIR}/gen")
  set_target_properties(vgpucupti PROPERTIES
    OUTPUT_NAME cupti SOVERSION ${SO_CUPTI} VERSION ${SO_CUPTI}.0)
  target_link_options(vgpucupti PRIVATE "-Wl,--version-script=${_cupti_map}")
  add_custom_command(TARGET vgpucupti POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcupti.so.${SO_CUPTI}.0"
            "${CMAKE_BINARY_DIR}/shim/libcupti.so.${SO_CUPTI}"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcupti.so.${SO_CUPTI}.0"
            "${CMAKE_BINARY_DIR}/shim/libcupti.so"
    COMMENT "Creating CUPTI shim symlinks (build/shim/libcupti.so.${SO_CUPTI})")

  # ---------------------------------------------------------------------------
  # libvgpucublas: cuBLAS, presented as libcublas.so.13. cuBLAS is a library,
  # not user code, so its arithmetic runs on the host at native speed rather
  # than through the SIMT interpreter -- see nvidia/docs/cublas.md.

    COMMENT "Creating cuBLAS shim symlinks (build/shim/libcublas.so.${SO_CUBLAS})")

  # ---------------------------------------------------------------------------
  # Additional vendor libraries. Same boundary as cuBLAS: library math runs on
  # the host, application kernels are simulated. Each is presented under the
  # real soname so unmodified programs link against it.
  # ---------------------------------------------------------------------------
  # The second field is only the fallback: the real major is read off the
  # installed toolkit, because it moves between CUDA releases (cuFFT is .so.11
  # under CUDA 12 and .so.12 under CUDA 13).
  set(_vgpu_libs
    "curand:10:curand_api"
    "cublasLt:13:cublaslt_api"
    "cufft:12:cufft_api"
    "cusparse:12:cusparse_api"

    string(REPLACE ":" ";" _parts "${_entry}")
    list(GET _parts 0 _lib)
    list(GET _parts 1 _fallback)

    vgpu_toolkit_soversion(${_lib} ${_fallback} _ver)
    vgpu_version_script(${_lib} ${_ver} _map)
    string(TOLOWER "${_lib}" _lower)
    # Some of these libraries also carry generated stubs for the entry points
    # they export but VirtualGPU does not implement: a framework resolves every
    # symbol at load time, so one missing name blocks the import outright.
    set(_extra_src "")
    if(EXISTS "${CMAKE_SOURCE_DIR}/nvidia/src/generated/${_lower}_stubs.cpp")
      set(_extra_src "nvidia/src/generated/${_lower}_stubs.cpp")
    endif()
    add_library(vgpu${_lower} SHARED nvidia/src/${_src}.cpp ${_extra_src})
    if(_lower STREQUAL "nvrtc")
      # A compiler, and nothing else: it touches no device, so it carries no
      # simulator and no runtime, as NVIDIA's has none. Linking libcudart here
      # put a second simulator into every driver-API program that uses NVRTC.
      target_link_libraries(vgpu${_lower} PRIVATE ${CMAKE_DL_LIBS})
    else()
      target_link_libraries(vgpu${_lower} PRIVATE vgpu vgpucudart)
    endif()
    target_include_directories(vgpu${_lower} PRIVATE ${CUDA_ABI_INCLUDE})
    set_target_properties(vgpu${_lower} PROPERTIES
      OUTPUT_NAME ${_lib} SOVERSION ${_ver} VERSION ${_ver}.0)
    target_link_options(vgpu${_lower} PRIVATE "-Wl,--version-script=${_map}")
    add_custom_command(TARGET vgpu${_lower} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
      COMMAND ${CMAKE_COMMAND} -E create_symlink "../lib${_lib}.so.${_ver}.0"
              "${CMAKE_BINARY_DIR}/shim/lib${_lib}.so.${_ver}"
      COMMAND ${CMAKE_COMMAND} -E create_symlink "../lib${_lib}.so.${_ver}.0"
              "${CMAKE_BINARY_DIR}/shim/lib${_lib}.so"
      COMMENT "Creating ${_lib} shim symlinks (soname lib${_lib}.so.${_ver})")
  endforeach()
  # cusolverSp is part of libcusolver and reads cuSPARSE's matrix descriptors
  # through cuSPARSE's getters, so like NVIDIA's libcusolver it links
  # libcusparse.
  target_sources(vgpucusolver PRIVATE nvidia/src/cusolver_sp.cpp nvidia/src/cusolver_rf.cpp)
  target_link_libraries(vgpucusolver PRIVATE vgpucusparse)
  # cusolverMg gathers a distributed matrix and runs cusolverDn on it.
  target_link_libraries(vgpucusolvermg PRIVATE vgpucusolver)
  # cuFFT's LTO callbacks: the callback image's PTX is read (fatbin.cpp) and
  # linked with the kernels that call it (ptx_link.cpp). Hidden, so these
  # copies stay this library's own rather than meeting libcudart's.
  target_sources(vgpucufft PRIVATE nvidia/src/fatbin.cpp nvidia/src/ptx_link.cpp)

  # libvgpunccl: NCCL, presented as libnccl.so.2. There is no NVLink to move
  # bytes over, so the transport is a directory of memory-mapped files -- which
  # is what lets one-rank-per-process launchers (torchrun) work unmodified.

  vgpu_forget_stale_path(NCCL_INCLUDE nccl.h)

  # libvgpunvjitlink and libvgpunvfatbin: nvJitLink and nvFatbin, presented as

  # ---------------------------------------------------------------------------
  # libvgpunvshmem: NVSHMEM's host library, presented as libnvshmem_host.so.3.
  # Each PE is a process with a simulated GPU; the symmetric heaps are shared
  # through the simulator's CUDA IPC, so every PE reaches every other by load
  # and store, and kernels built with NVIDIA's NVSHMEM headers and device
  # library run their inline device API against the state this library fills
  # in (nvidia/src/nvshmem_api.cpp, nvshmem_abi.hpp). The host API's
  # declarations are the simulator's own (nvidia/include/vgpu_nvshmem.h).
  # NVIDIA tags its symbols NVSHMEM; so does this, including the C++
  # nvshmemi_init_thread NVSHMEM's inline nvshmem_init() calls.
  # ---------------------------------------------------------------------------
  set(_nvshmem_map "${CMAKE_BINARY_DIR}/maps/libnvshmem_host.so.3.map")
  file(WRITE "${_nvshmem_map}"
       "NVSHMEM {\n  global: nvshmem*;\n  extern \"C++\" { nvshmemi_init_thread*; };\n  local: *;\n};\n")
  add_library(vgpunvshmem SHARED nvidia/src/nvshmem_api.cpp)
  target_link_libraries(vgpunvshmem PRIVATE vgpucudart ${CMAKE_DL_LIBS})
  target_include_directories(vgpunvshmem PRIVATE ${CUDA_ABI_INCLUDE} include)
  set_target_properties(vgpunvshmem PROPERTIES OUTPUT_NAME nvshmem_host SOVERSION 3 VERSION 3.8)
  target_link_options(vgpunvshmem PRIVATE "-Wl,--version-script=${_nvshmem_map}")
  add_custom_command(TARGET vgpunvshmem POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libnvshmem_host.so.3.8"
            "${CMAKE_BINARY_DIR}/shim/libnvshmem_host.so.3"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libnvshmem_host.so.3.8"
            "${CMAKE_BINARY_DIR}/shim/libnvshmem_host.so"
    COMMENT "Creating NVSHMEM shim symlinks (build/shim/libnvshmem_host.so.3)")

  # ---------------------------------------------------------------------------
  # libvgpucusparselt: cuSPARSELt, matrix products with a 2:4 structured sparse
  # operand, presented as libcusparseLt.so.0. Like cuDSS, NVIDIA's carries a
  # statically linked CUDA runtime that cannot reach a simulated driver, so the
  # simulator answers the documented API itself, from its own declarations
  # (nvidia/include/vgpu_cusparselt.h), and multiplies on the host. NVIDIA's
  # exports its symbols with no version tag, so these carry none.
  # ---------------------------------------------------------------------------
  set(_cusparselt_map "${CMAKE_BINARY_DIR}/maps/libcusparseLt.so.0.map")
  file(WRITE "${_cusparselt_map}" "{\n  global: cusparseLt*;\n  local: *;\n};\n")
  add_library(vgpucusparselt SHARED nvidia/src/cusparselt_api.cpp)
  target_link_libraries(vgpucusparselt PRIVATE vgpucudart)
  target_include_directories(vgpucusparselt PRIVATE ${CUDA_ABI_INCLUDE} include)
  set_target_properties(vgpucusparselt PROPERTIES OUTPUT_NAME cusparseLt SOVERSION 0 VERSION 0.10)
  target_link_options(vgpucusparselt PRIVATE "-Wl,--version-script=${_cusparselt_map}")
  add_custom_command(TARGET vgpucusparselt POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcusparseLt.so.0.10"
            "${CMAKE_BINARY_DIR}/shim/libcusparseLt.so.0"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcusparseLt.so.0.10"
            "${CMAKE_BINARY_DIR}/shim/libcusparseLt.so"
    COMMENT "Creating cuSPARSELt shim symlinks (build/shim/libcusparseLt.so.0)")
  # libvgpucutensor: cuTENSOR 2, presented as libcutensor.so.2. Like cuDSS,
  # NVIDIA's carries a statically linked CUDA runtime that cannot reach a
  # simulated driver, so the simulator answers the documented API itself, from
  # its own declarations (nvidia/include/vgpu_cutensor.h), computing on the
  # host. NVIDIA's tags each symbol with the release that introduced it
  # (libcutensor.1.0.0 ... libcutensor.2.5.0, a chain of versions); the map
  # reproduces the chain so a program linked against either library binds to
  # the other.
  # ---------------------------------------------------------------------------
  set(_cutensor_map "${CMAKE_BINARY_DIR}/maps/libcutensor.so.2.map")
  file(WRITE "${_cutensor_map}" [=[
libcutensor.1.0.0 {
  global: cutensorGetCudartVersion; cutensorGetErrorString; cutensorGetVersion;
  local: *;
};
libcutensor.1.1.0 { } libcutensor.1.0.0;
libcutensor.1.2.0 { } libcutensor.1.1.0;
libcutensor.1.3.0 {
  global: cutensorLoggerForceDisable; cutensorLoggerOpenFile; cutensorLoggerSetCallback;
          cutensorLoggerSetFile; cutensorLoggerSetLevel; cutensorLoggerSetMask;
} libcutensor.1.2.0;
libcutensor.1.6.0 { } libcutensor.1.3.0;
libcutensor.1.7.0 { global: cutensorCreate; cutensorDestroy; } libcutensor.1.6.0;
libcutensor.2.0.0 {
  global: CUTENSOR_COMPUTE_DESC_16F; CUTENSOR_COMPUTE_DESC_16BF; CUTENSOR_COMPUTE_DESC_TF32;
          CUTENSOR_COMPUTE_DESC_3XTF32; CUTENSOR_COMPUTE_DESC_32F; CUTENSOR_COMPUTE_DESC_64F;
          cutensorComputeDescriptorGetAttribute; cutensorComputeDescriptorSetAttribute; cutensorContract;
          cutensorCreateComputeDescriptor; cutensorCreateContraction; cutensorCreateElementwiseBinary;
          cutensorCreateElementwiseTrinary; cutensorCreatePermutation; cutensorCreatePlan;
          cutensorCreatePlanPreference; cutensorCreateReduction; cutensorCreateTensorDescriptor;
          cutensorDestroyComputeDescriptor; cutensorDestroyOperationDescriptor; cutensorDestroyPlan;
          cutensorDestroyPlanPreference; cutensorDestroyTensorDescriptor; cutensorElementwiseBinaryExecute;
          cutensorElementwiseTrinaryExecute; cutensorEstimateWorkspaceSize; cutensorHandleReadPlanCacheFromFile;
          cutensorHandleResizePlanCache; cutensorHandleWritePlanCacheToFile;
          cutensorOperationDescriptorGetAttribute; cutensorOperationDescriptorSetAttribute;
          cutensorOperationEstimateRuntime; cutensorOperationNumAlgos; cutensorPermute; cutensorPlanGetAttribute;
          cutensorPlanPreferenceSetAttribute; cutensorReadKernelCacheFromFile; cutensorReduce;
          cutensorTensorDescriptorSetAttribute; cutensorWriteKernelCacheToFile;
} libcutensor.1.7.0;
libcutensor.2.1.1 { global: cutensorContractTrinary; cutensorCreateContractionTrinary; } libcutensor.2.0.0;
libcutensor.2.2.1 { global: CUTENSOR_COMPUTE_DESC_9X16BF; };
libcutensor.2.3.0 {
  global: cutensorBlockSparseContract; cutensorCreateBlockSparseContraction;
          cutensorCreateBlockSparseTensorDescriptor; cutensorDestroyBlockSparseTensorDescriptor;
} libcutensor.2.2.1;
libcutensor.2.4.0 { global: CUTENSOR_COMPUTE_DESC_8XINT8; cutensorPlanPreferenceGetAttribute; } libcutensor.2.3.0;
libcutensor.2.5.0 { global: CUTENSOR_COMPUTE_DESC_4X16F; } libcutensor.2.4.0;
libcutensor.2.7.0 {
  global: cutensorCreateExtraction; cutensorCreateInsertion; cutensorExtract; cutensorInsert;
} libcutensor.2.5.0;
]=])
  add_library(vgpucutensor SHARED nvidia/src/cutensor_api.cpp)
  target_link_libraries(vgpucutensor PRIVATE vgpucudart)
  target_include_directories(vgpucutensor PRIVATE ${CUDA_ABI_INCLUDE} include)
  set_target_properties(vgpucutensor PROPERTIES OUTPUT_NAME cutensor SOVERSION 2 VERSION 2.8)
  target_link_options(vgpucutensor PRIVATE "-Wl,--version-script=${_cutensor_map}")
  add_custom_command(TARGET vgpucutensor POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcutensor.so.2.8"
            "${CMAKE_BINARY_DIR}/shim/libcutensor.so.2"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcutensor.so.2.8"
            "${CMAKE_BINARY_DIR}/shim/libcutensor.so"
    COMMENT "Creating cuTENSOR shim symlinks (build/shim/libcutensor.so.2)")

  # ---------------------------------------------------------------------------
  # libvgpucutensornet: cuTensorNet (cuQuantum), presented as
  # libcutensornet.so.2, on the simulator's cuTENSOR and cuSOLVER as NVIDIA's
  # is on theirs. NVIDIA's tags its symbols libcutensornet.so.0.
  # ---------------------------------------------------------------------------
  vgpu_version_script(cutensornet 0 _cutensornet_map)
  add_library(vgpucutensornet SHARED nvidia/src/cutensornet_api.cpp)
  target_link_libraries(vgpucutensornet PRIVATE vgpucutensor vgpucusolver vgpucudart)
  target_include_directories(vgpucutensornet PRIVATE ${CUDA_ABI_INCLUDE} include)
  set_target_properties(vgpucutensornet PROPERTIES OUTPUT_NAME cutensornet SOVERSION 2 VERSION 2.14)
  target_link_options(vgpucutensornet PRIVATE "-Wl,--version-script=${_cutensornet_map}")
  add_custom_command(TARGET vgpucutensornet POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcutensornet.so.2.14"
            "${CMAKE_BINARY_DIR}/shim/libcutensornet.so.2"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcutensornet.so.2.14"
            "${CMAKE_BINARY_DIR}/shim/libcutensornet.so"
    COMMENT "Creating cuTensorNet shim symlinks (build/shim/libcutensornet.so.2)")

  # ---------------------------------------------------------------------------
  # libvgpucudnn: cuDNN, presented as libcudnn.so.9. Built against the real
  # cuDNN API headers -- hand-writing a vendor struct layout has silently
  # produced garbage twice, so we never do it. The headers are NVIDIA's and are
  # not kept in this tree: CUDNN_HEADER_DIR (environment) names a directory that
  # has them, and otherwise they are fetched into the build directory from the
  # nvidia-cudnn-cu12 wheel on PyPI (scripts/fetch-cudnn-headers.py: about 1 MB,
  # each file checked against scripts/cudnn-headers.sha256). Without network
  # access, or with VGPU_FETCH_CUDNN_HEADERS=OFF, libvgpucudnn is skipped.
  # Real cuDNN exports versioned symbols tagged "libcudnn.so.9"; the version
  # script gives ours the same tag so the loader binds without the
  # "no version information available" warning.
  # ---------------------------------------------------------------------------
  option(VGPU_FETCH_CUDNN_HEADERS "Fetch cuDNN's API headers from PyPI when none are given" ON)
  vgpu_forget_stale_path(CUDNN_INCLUDE cudnn.h)
  find_path(CUDNN_INCLUDE cudnn.h
    PATHS ENV CUDNN_HEADER_DIR "${CMAKE_BINARY_DIR}/cudnn_include"
    NO_DEFAULT_PATH)
  if(NOT CUDNN_INCLUDE AND VGPU_FETCH_CUDNN_HEADERS)
    find_program(VGPU_PYTHON NAMES python3 python)
    if(VGPU_PYTHON)
      execute_process(
        COMMAND "${VGPU_PYTHON}" "${CMAKE_SOURCE_DIR}/scripts/fetch-cudnn-headers.py" "${CMAKE_BINARY_DIR}/cudnn_include"
        RESULT_VARIABLE _cudnn_fetch_rc OUTPUT_VARIABLE _cudnn_fetch_out ERROR_VARIABLE _cudnn_fetch_err)
      if(_cudnn_fetch_rc EQUAL 0)
        message(STATUS "${_cudnn_fetch_out}")
        unset(CUDNN_INCLUDE CACHE)
        find_path(CUDNN_INCLUDE cudnn.h PATHS "${CMAKE_BINARY_DIR}/cudnn_include" NO_DEFAULT_PATH)
      else()
        message(STATUS "cuDNN headers could not be fetched: ${_cudnn_fetch_err}")
      endif()
    endif()
  endif()
  if(CUDNN_INCLUDE)
    message(STATUS "cuDNN headers at ${CUDNN_INCLUDE}: building libvgpucudnn")
    # Every entry point PyTorch binds is implemented, so cuDNN has no
    # generated stubs (nvidia/src/generated/) any more.
    add_library(vgpucudnn SHARED nvidia/src/cudnn_api.cpp nvidia/src/cudnn_backend.cpp nvidia/src/cudnn_rnn.cpp
      nvidia/src/cudnn_common.cpp)
    target_link_libraries(vgpucudnn PRIVATE vgpu vgpucudart)
    target_include_directories(vgpucudnn PRIVATE ${CUDNN_INCLUDE} ${CUDA_ABI_INCLUDE})
    set_target_properties(vgpucudnn PROPERTIES OUTPUT_NAME cudnn SOVERSION 9 VERSION 9.0)
    target_link_options(vgpucudnn PRIVATE
      "-Wl,--version-script=${CMAKE_SOURCE_DIR}/nvidia/src/cudnn.map")
    add_custom_command(TARGET vgpucudnn POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
      COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcudnn.so.9.0"
              "${CMAKE_BINARY_DIR}/shim/libcudnn.so.9"
      COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcudnn.so.9.0"
              "${CMAKE_BINARY_DIR}/shim/libcudnn.so"
      COMMENT "Creating cuDNN shim symlinks (build/shim/libcudnn.so.9)")
  else()
    message(STATUS "cuDNN headers not found (set CUDNN_HEADER_DIR, or allow VGPU_FETCH_CUDNN_HEADERS): skipping libvgpucudnn")
  endif()

  # ---------------------------------------------------------------------------
  # libvgpunvenc: VirtualGPU's NVENC implementation, presented as
  # libnvidia-encode.so.1. Applications dlopen that bare soname, which searches
  # LD_LIBRARY_PATH first, so the shim directory wins over the real driver
  # library. Needs the Video Codec SDK header, vendored in
  # nvidia/third_party/nvenc_include so the shim builds from this tree alone
  # (a container build sees nothing beside it).
  # ---------------------------------------------------------------------------
  vgpu_forget_stale_path(NVENC_INCLUDE nvEncodeAPI.h)
  find_path(NVENC_INCLUDE nvEncodeAPI.h
    PATHS "${CMAKE_SOURCE_DIR}/nvidia/third_party/nvenc_include"
          "${CMAKE_SOURCE_DIR}/../pantheongpu/kernels/common/nvenc"
          ENV NVENC_HEADER_DIR)
  if(NVENC_INCLUDE)
    message(STATUS "NVENC header at ${NVENC_INCLUDE}: building libvgpunvenc")
    add_library(vgpunvenc SHARED nvidia/src/nvenc_api.cpp)
    target_link_libraries(vgpunvenc PRIVATE vgpu vgpucudart)
    target_include_directories(vgpunvenc PRIVATE ${NVENC_INCLUDE})
    set_target_properties(vgpunvenc PROPERTIES OUTPUT_NAME nvidia-encode SOVERSION 1)
    add_custom_command(TARGET vgpunvenc POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
      COMMAND ${CMAKE_COMMAND} -E create_symlink "../libnvidia-encode.so.1"
              "${CMAKE_BINARY_DIR}/shim/libnvidia-encode.so.1"
      COMMENT "Creating NVENC shim symlink (build/shim/libnvidia-encode.so.1)")
  else()
    message(STATUS "nvEncodeAPI.h not found: skipping libvgpunvenc")
  endif()
else()
  message(STATUS "CUDA ABI headers not found: skipping libvgpucudart (core unaffected)")
endif()

# vgpu_unit_test(NAME [SOURCE]): tests/unit/<name>.cpp, or a source of its own
# for a vendor's tests (amd/tests/unit, nvidia/tests/unit).
function(vgpu_unit_test name)
  set(_src tests/unit/${name}.cpp)
  if(ARGC GREATER 1)
    set(_src ${ARGV1})
  endif()
  add_executable(${name} ${_src})
  target_link_libraries(${name} PRIVATE vgpu)
  target_include_directories(${name} PRIVATE tests/framework)
  add_test(NAME ${name} COMMAND ${name})
endfunction()
