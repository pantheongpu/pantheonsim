cmake_minimum_required(VERSION 3.20)
project(pantheonsim VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)

# ---------------------------------------------------------------------------
# Embed device profiles (profiles/*.yaml) into a generated header so binaries
# are self-contained. Re-runs automatically when a profile file changes.
# ---------------------------------------------------------------------------
set(VGPU_PROFILES
  nvidia/a10
  nvidia/a100
  nvidia/h100
  nvidia/h200
  nvidia/b200
)
set(_gen_dir "${CMAKE_BINARY_DIR}/generated")
set(_gen "${_gen_dir}/vgpu/embedded_profiles.hpp")
set(_content "// Generated by CMake from profiles/ — do not edit.\n#pragma once\n\nnamespace vgpu::embedded {\n\nstruct EmbeddedProfile {\n  const char* id;\n  const char* yaml;\n};\n\ninline constexpr EmbeddedProfile kProfiles[] = {\n")
foreach(_p IN LISTS VGPU_PROFILES)
  file(READ "${CMAKE_SOURCE_DIR}/profiles/${_p}.yaml" _yaml)
  string(APPEND _content "  {\"${_p}\", R\"VGPUPROFILE(${_yaml})VGPUPROFILE\"},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "profiles/${_p}.yaml")
endforeach()
string(APPEND _content "};\n\n}  // namespace vgpu::embedded\n")
file(WRITE "${_gen}" "${_content}")

# ---------------------------------------------------------------------------
# Core static library: profiles, memory, PTX, execution engine, runtime.
# ---------------------------------------------------------------------------
add_library(vgpu STATIC
  src/core/error.cpp
  src/core/yamlish.cpp
  src/core/profile.cpp
  src/core/registry.cpp
  src/core/memory.cpp
  src/ptx/ast.cpp
  src/ptx/lexer.cpp
  src/ptx/parser.cpp
  src/exec/scheduler.cpp
  src/exec/interpreter.cpp
  src/runtime/runtime.cpp
)
target_include_directories(vgpu PUBLIC include "${_gen_dir}")

# libvgpucuda: the CUDA Driver API shim external programs link against.
add_library(vgpucuda SHARED src/cuda/driver_api.cpp)
target_link_libraries(vgpucuda PRIVATE vgpu)

# CLI
add_executable(vgpu_cli src/cli/main.cpp src/cli/demo.cpp)
target_link_libraries(vgpu_cli PRIVATE vgpu)
set_target_properties(vgpu_cli PROPERTIES OUTPUT_NAME vgpu)

# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------
enable_testing()
function(vgpu_unit_test name)
  add_executable(${name} tests/unit/${name}.cpp)
  target_link_libraries(${name} PRIVATE vgpu)
  target_include_directories(${name} PRIVATE tests/framework)
  add_test(NAME ${name} COMMAND ${name})
endfunction()

vgpu_unit_test(test_yamlish)
vgpu_unit_test(test_profile)
vgpu_unit_test(test_memory)
vgpu_unit_test(test_ptx)
target_compile_definitions(test_ptx PRIVATE VGPU_KERNEL_DIR="${CMAKE_SOURCE_DIR}/tests/kernels")
vgpu_unit_test(test_exec)
target_compile_definitions(test_exec PRIVATE VGPU_KERNEL_DIR="${CMAKE_SOURCE_DIR}/tests/kernels")
vgpu_unit_test(test_runtime)

# The C harness: an "external application" built with a plain C compiler
# against the documented ABI only (vgpu_cuda.h + libvgpucuda.so).
add_executable(vector_add_driver tests/c_harness/vector_add_driver.c)
target_link_libraries(vector_add_driver PRIVATE vgpucuda)
target_include_directories(vector_add_driver PRIVATE include)
add_test(NAME c_harness_vector_add COMMAND vector_add_driver)
set_tests_properties(c_harness_vector_add PROPERTIES ENVIRONMENT "VGPU_GPU=nvidia/h100")
