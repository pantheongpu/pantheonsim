# ---------------------------------------------------------------------------
# Embed device profiles (profiles/*.yaml) into a generated header so binaries
# are self-contained. Re-runs automatically when a profile file changes.
# ---------------------------------------------------------------------------
set(VGPU_PROFILES
  nvidia/a10
  nvidia/a100
  nvidia/h100
  nvidia/h200
  nvidia/b200
  amd/mi300x
  amd/mi325x
  amd/mi350x
)
set(_gen_dir "${CMAKE_BINARY_DIR}/generated")
set(_gen "${_gen_dir}/vgpu/embedded_profiles.hpp")
set(_content "// Generated by CMake from profiles/ — do not edit.\n#pragma once\n\nnamespace vgpu::embedded {\n\nstruct EmbeddedProfile {\n  const char* id;\n  const char* yaml;\n};\n\ninline constexpr EmbeddedProfile kProfiles[] = {\n")
foreach(_p IN LISTS VGPU_PROFILES)
  file(READ "${CMAKE_SOURCE_DIR}/profiles/${_p}.yaml" _yaml)
  string(APPEND _content "  {\"${_p}\", R\"VGPUPROFILE(${_yaml})VGPUPROFILE\"},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "profiles/${_p}.yaml")
endforeach()
string(APPEND _content "};\n\n}  // namespace vgpu::embedded\n")
file(WRITE "${_gen}" "${_content}")

# libvgpucuda: the CUDA Driver API shim external programs link against.
add_library(vgpucuda SHARED src/cuda/driver_api.cpp src/cuda/fatbin.cpp)
target_link_libraries(vgpucuda PRIVATE vgpu ${CMAKE_DL_LIBS})
# Present the real driver's identity: apps linked with -lcuda record SONAME
# "libcuda.so.1" and resolve to this library through LD_LIBRARY_PATH at run time.
set_target_properties(vgpucuda PROPERTIES OUTPUT_NAME cuda SOVERSION 1 VERSION 1.1)
# The shim directory: prepend to LD_LIBRARY_PATH and unmodified CUDA binaries
# (static cudart included) dlopen VirtualGPU as the driver.
add_custom_command(TARGET vgpucuda POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
  COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcuda.so.1.1" "${CMAKE_BINARY_DIR}/shim/libcuda.so.1"
  COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcuda.so.1.1" "${CMAKE_BINARY_DIR}/shim/libcuda.so"
  COMMENT "Creating driver shim symlinks (build/shim/libcuda.so.1)")

# ---------------------------------------------------------------------------
# libvgpucudart: the CUDA *runtime* API shim. Lets unmodified nvcc-compiled
# applications (built against shared cudart) run on VirtualGPU. Requires the
# CUDA toolkit's ABI headers at build time (driver_types.h) for exact struct
# layout; built only when they are found, so the core stays dependency-free.
# ---------------------------------------------------------------------------
# Derive the ABI include from the nvcc that will compile the workloads, so the
# shim's cudaDeviceProp layout matches the application's exactly. A mismatched
# system copy (e.g. an older /usr/include/driver_types.h) would corrupt caller
# structs, so nvcc's own toolkit include is the only trustworthy source.
find_program(VGPU_NVCC nvcc)
if(VGPU_NVCC)
  get_filename_component(_nvcc_bin "${VGPU_NVCC}" DIRECTORY)
  get_filename_component(_cuda_root "${_nvcc_bin}" DIRECTORY)
  find_path(CUDA_ABI_INCLUDE driver_types.h
    PATHS "${_cuda_root}/include" "${_cuda_root}/targets/x86_64-linux/include"
    NO_DEFAULT_PATH)
endif()
if(NOT CUDA_ABI_INCLUDE)
  find_path(CUDA_ABI_INCLUDE driver_types.h
    PATHS /usr/local/cuda/include /opt/cuda/include ENV CUDA_HOME ENV CUDA_PATH
    PATH_SUFFIXES include NO_DEFAULT_PATH)
endif()
if(CUDA_ABI_INCLUDE)
  message(STATUS "CUDA ABI headers at ${CUDA_ABI_INCLUDE}: building libvgpucudart")
  add_library(vgpucudart SHARED src/cuda/runtime_api.cpp src/cuda/fatbin.cpp)
  target_link_libraries(vgpucudart PRIVATE vgpu ${CMAKE_DL_LIBS})
  target_include_directories(vgpucudart PRIVATE ${CUDA_ABI_INCLUDE} src/cuda)
  # Present as libcudart.so.13 so unmodified binaries find us via LD_LIBRARY_PATH.
  # Same identity trick as libcuda: become libcudart.so.13 with that SONAME.
  set_target_properties(vgpucudart PROPERTIES OUTPUT_NAME cudart SOVERSION 13 VERSION 13.0)
  target_link_options(vgpucudart PRIVATE
    "-Wl,--version-script=${CMAKE_SOURCE_DIR}/src/cuda/cudart.map")
  add_custom_command(TARGET vgpucudart POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcudart.so.13.0" "${CMAKE_BINARY_DIR}/shim/libcudart.so.13"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcudart.so.13.0" "${CMAKE_BINARY_DIR}/shim/libcudart.so.12"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libcudart.so.13.0" "${CMAKE_BINARY_DIR}/shim/libcudart.so"
    COMMENT "Creating runtime shim symlinks (build/shim/libcudart.so.13)")
  # ---------------------------------------------------------------------------
  # libvgpunvml: NVML, presented as libnvidia-ml.so.1. nvidia-smi dlopens that
  # soname and resolves entries by name, so the real nvidia-smi binary reports
  # VirtualGPU's devices when the shim directory is on LD_LIBRARY_PATH.
  # ---------------------------------------------------------------------------
  add_library(vgpunvml SHARED src/cuda/nvml_api.cpp)
  target_link_libraries(vgpunvml PRIVATE vgpu)
  target_include_directories(vgpunvml PRIVATE ${CUDA_ABI_INCLUDE})
  set_target_properties(vgpunvml PROPERTIES OUTPUT_NAME nvidia-ml SOVERSION 1)
  add_custom_command(TARGET vgpunvml POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libnvidia-ml.so.1"
            "${CMAKE_BINARY_DIR}/shim/libnvidia-ml.so.1"
    COMMAND ${CMAKE_COMMAND} -E create_symlink "../libnvidia-ml.so.1"
            "${CMAKE_BINARY_DIR}/shim/libnvidia-ml.so"
    COMMENT "Creating NVML shim symlinks (build/shim/libnvidia-ml.so.1)")

  # ---------------------------------------------------------------------------
  # libvgpunvenc: VirtualGPU's NVENC implementation, presented as
  # libnvidia-encode.so.1. Applications dlopen that bare soname, which searches
  # LD_LIBRARY_PATH first, so the shim directory wins over the real driver
  # library. Needs the Video Codec SDK header (vendored by the workload tree).
  # ---------------------------------------------------------------------------
  find_path(NVENC_INCLUDE nvEncodeAPI.h
    PATHS "${CMAKE_SOURCE_DIR}/../pantheongpu/kernels/common/nvenc"
          ENV NVENC_HEADER_DIR)
  if(NVENC_INCLUDE)
    message(STATUS "NVENC header at ${NVENC_INCLUDE}: building libvgpunvenc")
    add_library(vgpunvenc SHARED src/cuda/nvenc_api.cpp)
    target_link_libraries(vgpunvenc PRIVATE vgpu vgpucudart)
    target_include_directories(vgpunvenc PRIVATE ${NVENC_INCLUDE})
    set_target_properties(vgpunvenc PROPERTIES OUTPUT_NAME nvidia-encode SOVERSION 1)
    add_custom_command(TARGET vgpunvenc POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shim"
      COMMAND ${CMAKE_COMMAND} -E create_symlink "../libnvidia-encode.so.1"
              "${CMAKE_BINARY_DIR}/shim/libnvidia-encode.so.1"
      COMMENT "Creating NVENC shim symlink (build/shim/libnvidia-encode.so.1)")
  else()
    message(STATUS "nvEncodeAPI.h not found: skipping libvgpunvenc")
  endif()
else()
  message(STATUS "CUDA ABI headers not found: skipping libvgpucudart (core unaffected)")
endif()

# CLI
add_executable(vgpu_cli src/cli/main.cpp src/cli/demo.cpp src/cli/smi.cpp src/cli/serve.cpp)
target_link_libraries(vgpu_cli PRIVATE vgpu)
set_target_properties(vgpu_cli PROPERTIES OUTPUT_NAME vgpu)
# Drop-in monitoring commands. On a CPU-only machine nvidia-smi / rocm-smi /
# rocm_agent_enumerator do not exist at all, so VirtualGPU supplies them; they
# render the same telemetry the NVML shim serves.
add_custom_command(TARGET vgpu_cli POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/bin"
  COMMAND ${CMAKE_COMMAND} -E create_symlink "../vgpu" "${CMAKE_BINARY_DIR}/bin/vgpu"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/tools/nvidia-smi" "${CMAKE_BINARY_DIR}/bin/nvidia-smi"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/tools/rocm-smi" "${CMAKE_BINARY_DIR}/bin/rocm-smi"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/tools/rocm_agent_enumerator" "${CMAKE_BINARY_DIR}/bin/rocm_agent_enumerator"
  COMMENT "Installing drop-in monitoring tools into build/bin")

enable_testing()
function(vgpu_unit_test name)
  add_executable(${name} tests/unit/${name}.cpp)
  target_link_libraries(${name} PRIVATE vgpu)
  target_include_directories(${name} PRIVATE tests/framework)
  add_test(NAME ${name} COMMAND ${name})
endfunction()
